-- Aggregate 420M user interaction telemetry records in milliseconds
SELECT
  event_category,
  count(*) AS total_events,
  quantile(0.99)(query_duration_ms) AS p99_latency_ms
FROM telemetry.event_stream
WHERE event_date = today()
GROUP BY event_category
ORDER BY total_events DESC LIMIT 3;
